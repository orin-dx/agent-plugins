fn git_evidence(workspace: &Utf8Path) -> GitEvidence {
    let head = Command::new("git")
        .arg("-C")
        .arg(workspace)
        .args(["rev-parse", "HEAD"])
        .output();
    let Ok(head) = head else {
        return GitEvidence::Unavailable {
            reason: "git_unavailable".to_owned(),
        };
    };
    if !head.status.success() {
        return GitEvidence::Unavailable {
            reason: "not_a_git_worktree".to_owned(),
        };
    }
    let status = Command::new("git")
        .arg("-C")
        .arg(workspace)
        .args(["status", "--porcelain=v1"])
        .output();
    let Ok(status) = status else {
        return GitEvidence::Unavailable {
            reason: "git_status_unavailable".to_owned(),
        };
    };
    if !status.status.success() {
        return GitEvidence::Unavailable {
            reason: "git_status_unavailable".to_owned(),
        };
    }
    GitEvidence::Available {
        head: String::from_utf8_lossy(&head.stdout).trim().to_owned(),
        dirty: !status.stdout.is_empty(),
    }
}
